\chapter{Theoretical Background}

This chapter establishes the theoretical and mathematical foundations directly supporting our four research pillars: bounded LLM agent routing, supporting speech pre-processing, censored demand recovery, and profit-gated perishable inventory optimization.

\section{Fundamental Concepts}

\subsection*{Bounded LLM Agents and Zero-Planner Hybrid Routing}
An \textbf{AI agent} augments a Large Language Model (LLM) with structured perception, session memory, typed external tools, and explicit policy control~\cite{yao2023react,schick2023toolformer}. In retail operations, open-ended ReAct loops suffer from cumulative planning latency and factual drift on prices or stock. Extending adaptive query-routing principles~\cite{chen2023frugalgpt,ong2024routellm}, our architecture enforces \textbf{deterministic-first, LLM-bounded routing}:
\begin{itemize}
    \item \textbf{Zero-Planner Hybrid Routing (\texttt{Retail-AI-}):} Every customer turn passes through sub-3\,ms deterministic gates (medical/medication refusal, out-of-catalog filtering, OWASP prompt-injection defense). Standard catalog requests (pricing, unit-price comparison, promotions, stock, synchronized 30-day bestsellers) execute directly via inverted-index search and deterministic calculators before streaming concise explanations, bypassing multi-step JSON planner loops; only high-risk clinical/allergen or unresolved multi-constraint queries escalate to bounded structured planning.
    \item \textbf{Read/Proposal Separation and HITL Governance (\texttt{ops\_chest}):} The store-operations agent accesses 22 read/UI-form tools and 9 proposal tools, with zero direct-execution tools in the LLM schema. Every operational change traverses a persistent proposal state machine (\texttt{PENDING} $\to$ \texttt{APPROVED}/\texttt{REJECTED} $\to$ \texttt{EXECUTED}) triggered strictly by authenticated manager approval.
\end{itemize}

\subsection*{Supporting Speech Processing Front-End}
To support hands-free Vietnamese voice queries at the sales counter, \texttt{Retail-AI-} incorporates a modular audio front-end combining recurrent voice activity detection (\emph{Silero VAD~v6}~\cite{silero_vad} on 32\,ms frames at 16\,kHz), high-pass filtering with neural speech enhancement (e.g., \texttt{DPDFNet2} from the DeepFilterNet family~\cite{schroter2022deepfilternet,sherpa_onnx}), streaming ASR with domain phrase biasing (\code{PhraseListGrammar}~\cite{azure_speech}), and clause-level TTS. Software implementation details (browser edge denoising, AEC preservation, and deployment configs) are deferred to Appendix~A.

\subsection*{Censored Demand Recovery, Quantile Forecasting, and Profit-Gated Routing}
Short-shelf-life perishables (1--3 days) exhibit three defining analytical properties:
\begin{itemize}
    \item \textbf{Right-Censored Demand:} Whenever shelves empty before closing, observed sales underestimate true demand~\cite{wecker1978stockouts,vulcano2012primary}. Using empirical hourly sales-share profiles learned on fully stocked days of FreshRetailNet-50K~\cite{freshretailnet50k}, daily demand is uncensored prior to feature generation.
    \item \textbf{Demand Taxonomy and LightGBM vs.\ EWMA:} Under the Syntetos--Boylan ADI/$\text{CV}^2$ taxonomy~\cite{syntetos2005intermittent}, store-SKU series span smooth, erratic, and lumpy regimes. \emph{LightGBM}~\cite{ke2017lightgbm} learns nonlinear lag, calendar, and stockout interactions under pinball loss at quantiles $q \in \{0.10, 0.50, 0.90\}$, whereas seasonal \emph{EWMA} with a Poisson variance floor supplies a low-variance statistical estimator for noisy small-store segments.
    \item \textbf{Decision-Focused Routing (Predict-then-Optimize):} Consistent with Ban--Rudin~\cite{ban2019bigdata} and Elmachtoub--Grigas~\cite{elmachtoub2022spo}, minimizing statistical error (WAPE) does not automatically maximize gross profit under asymmetric FEFO--Newsvendor costs~\cite{nahmias2011perishable,silver2017inventory}. \texttt{B-Gate} decouples point estimation ($\hat{q}_{50}^{\text{LGBM}}$) from dispersion estimation ($\sigma_{\text{EWMA}}$) and activates ML only on segments delivering strictly positive out-of-sample backtest profit gains.
\end{itemize}

\section{Mathematical Formulations}

\intuitionbox{Plain-Language Intuition Behind the Three Core Equations}{%
Before inspecting the formal equations below, three everyday retail examples clarify their purpose:
\begin{enumerate}[leftmargin=1.4em, itemsep=2pt, topsep=2pt]
    \item \textbf{Uncensoring Stockout Days (Eq.~\ref{eq:wape_recovery}):} Suppose fresh greens normally sell 50\% of daily volume before 14:00. If shelves empty at 14:00 after selling 10 packs, true unconstrained demand is estimated at $10 / 0.5 = 20$ packs rather than 10.
    \item \textbf{Asymmetric Perishable Newsvendor (Eq.~\ref{eq:newsvendor}--\ref{eq:order_qty}):} Unsold 1--3 day perishables forfeit 100\% of unit cost, whereas a stockout forfeits only gross margin. The critical fractile $SL^*$ balances both losses, while a 95\% floor protects traffic-driving staples.
    \item \textbf{Profit-Gated Routing \texttt{B-Gate} (Eq.~\ref{eq:profit_gate}):} Lower forecast WAPE may still underperform on gross profit if overly wide quantile spreads cause excess spoilage. \texttt{B-Gate} activates LightGBM strictly on store--SKU segments with positive out-of-sample profit lift over EWMA.
\end{enumerate}}

\subsection*{Supporting Audio Conditioning and ASR Metrics}
On the server denoising path, a zero-phase fourth-order frequency-domain high-pass filter ($f_c = 85\,\text{Hz}$) attenuates low-frequency hum:
\begin{equation}
    H(f) = \frac{1}{1 + \left(\frac{f_c}{\max(f,\,1)}\right)^4}, \qquad H(0) = 0, \qquad x_{\text{HPF}} = \mathcal{F}^{-1}\!\left\{\mathcal{F}\{x\} \cdot H(f)\right\}.
    \label{eq:hpf85}
\end{equation}
Clip SNR ($\widehat{\text{SNR}}_{\text{dB}}$) is estimated across 20\,ms frames ($N_f = 320$ samples at 16\,kHz) via the ratio of 88th-percentile active RMS to 15th-percentile floor RMS, controlling adaptive wet/dry mixing:
\begin{equation}
    \widehat{\text{SNR}}_{\text{dB}} = \operatorname{clip}\!\left(20 \log_{10}\!\left(\frac{P_{88}(\text{RMS}_{\text{frame}})}{\max(P_{15}(\text{RMS}_{\text{frame}}),\,10^{-6})}\right),\,-5,\,45\right),
    \label{eq:snr_est}
\end{equation}
\begin{equation}
    y[n] = w(\widehat{\text{SNR}}_{\text{dB}})\, x_{\text{DPDFNet2}}[n] + \bigl(1 - w(\widehat{\text{SNR}}_{\text{dB}})\bigr)\, x_{\text{HPF}}[n], \quad
    w = \begin{cases}
        0.00 & \widehat{\text{SNR}}_{\text{dB}} \ge 28\,\text{dB}, \\
        0.85 & 10 \le \widehat{\text{SNR}}_{\text{dB}} < 28\,\text{dB}, \\
        0.75 & \widehat{\text{SNR}}_{\text{dB}} < 10\,\text{dB}.
    \end{cases}
    \label{eq:snr_blend}
\end{equation}
Recognition accuracy is measured by Word Error Rate (WER), and end-to-end voice turn latency decomposes into five stages:
\begin{equation}
    \text{WER} = \frac{S + D + I}{N}, \qquad T_{\text{E2E}} = T_{\text{VAD}} + T_{\text{STT-post}} + T_{\text{retrieval}} + T_{\text{TTFT}} + T_{\text{TTS-1st}}.
    \label{eq:wer_latency}
\end{equation}

\subsection*{Unit-Price Normalization (\texttt{Retail-AI-})}
To compare heterogeneous package sizes fairly (e.g., per 100\,g, per kg, or per L), listed retail price $P_{\text{VND}}$ is normalized into canonical base-unit price and percentage discount:
\begin{equation}
    P_{\text{unit}} = \frac{P_{\text{VND}}}{Q_{\text{base}}}, \qquad d_{\%} = \max\!\left(0,\, \frac{P_{\text{regular}} - P_{\text{VND}}}{P_{\text{regular}}} \times 100\right).
    \label{eq:unit_price}
\end{equation}

\subsection*{Demand Recovery, Pinball Loss, Perishable Newsvendor--FEFO, and Profit Gate}
Let $s_{i,t}$ denote observed sales of SKU $i$ on day $t$ and $f_{i,t} \in (0,1]$ the cumulative empirical hourly demand share elapsed prior to stockout. Recovered demand $\hat{D}_{i,t}$, WAPE, and quantile pinball loss $\mathcal{L}_q$ are defined as:
\begin{equation}
    \hat{D}_{i,t} = \frac{s_{i,t}}{\max(f_{i,t},\,\epsilon)}, \qquad
    \text{WAPE} = \frac{\sum_{i,t} \left|D_{i,t} - \hat{y}_{i,t}\right|}{\sum_{i,t} \left|D_{i,t}\right|}, \qquad
    \mathcal{L}_q(y, \hat{y}) = \max\bigl(q(y-\hat{y}),\,(q-1)(y-\hat{y})\bigr).
    \label{eq:wape_recovery}
\end{equation}
Given selling price $p$, unit cost $c$, salvage value $v$, and stockout opportunity cost $s_{\text{out}}$, the optimal Newsvendor service level $SL^*$ and shelf-life-capped order quantity $Q^*$ (review period $R$, arrival-day FEFO unmet demand $U_R$, leftover $L_R$, sellable window $W = \min(90, L)$, available shelf capacity $C_{\text{room}}$) are:
\begin{equation}
    SL^* = \max\!\left(SL_{\text{floor}},\, \operatorname{clip}\!\left(\frac{p - c + s_{\text{out}}}{(p - c + s_{\text{out}}) + (c - v)},\,0.30,\,0.98\right)\right), \qquad z = \Phi^{-1}(SL^*),
    \label{eq:newsvendor}
\end{equation}
\begin{equation}
\begin{aligned}
    Q_{\text{raw}} &= U_R + z\,\sigma_{\text{daily}}\sqrt{R} - L_R, \qquad
    Q_{\text{cap}} = \max(0,\, D_W - S_W) + \max(z,0)\,\sigma_{\text{daily}}\sqrt{W}, \\
    Q^* &= \max\!\bigl(0,\,\min(Q_{\text{raw}}, Q_{\text{cap}}, C_{\text{room}})\bigr),
\end{aligned}
    \label{eq:order_qty}
\end{equation}
with $SL_{\text{floor}} \in \{0.95\text{ (traffic driver)},\,0.90\text{ (ABC-A)},\,0.50\text{ (other)}\}$. The profit gate (\texttt{B-Gate}) routes segment $g = (\text{sales tier} \times \text{intermittency} \times \text{store-size tier})$ to LightGBM if and only if its normalized store-profit gain over EWMA on training folds is strictly positive:
\begin{equation}
\begin{aligned}
    V(g) &= \sum_{s \in \mathcal{S}_{\text{train}}} \frac{\sum_{i \in g \cap s}\bigl(\Pi^{\text{ML}}_{i,s} - \Pi^{\text{EWMA}}_{i,s}\bigr)}{\Pi^{\text{EWMA}}_s} > 0, \\
    \Pi_{i,s} &= \text{Revenue}_{i,s} - \text{COGS}_{i,s} - \text{WasteCost}_{i,s}.
\end{aligned}
    \label{eq:profit_gate}
\end{equation}
