\hline
\textbf{ASR / STT} & Automatic Speech Recognition / Speech-to-Text -- Nhận dạng giọng nói sang văn bản \\
\hline
\textbf{BFCL} & Berkeley Function Calling Leaderboard -- Bộ kiểm chuẩn đánh giá năng lực gọi hàm của LLM \\
\hline
\textbf{CI} & Confidence Interval -- Khoảng tin cậy thống kê (ví dụ bootstrap 95\% CI) \\
\hline
\textbf{CLIP} & Contrastive Language--Image Pretraining -- Mô hình nền tảng thị giác--ngôn ngữ dùng làm giáo viên chưng cất \\
\hline
\textbf{CV / $\text{CV}^2$} & Coefficient of Variation / Cross-Validation -- Hệ số biến thiên / Kiểm định chéo \\
\hline
\textbf{DINOv2} & Self-Distilled Vision Transformer v2 -- Mô hình nền tảng trích xuất đặc trưng hình học đa góc chụp \\
\hline
\textbf{DPDFNet2} & Dual-Path Deep Filtering Network v2 -- Mô hình khử nhiễu giọng nói nơ-ron ONNX \\
\hline
\textbf{EAN-13} & European Article Number (13 chữ số) -- Mã vạch định danh sản phẩm bán lẻ \\
\hline
\textbf{EDA} & Exploratory Data Analysis -- Phân tích khám phá dữ liệu \\
\hline
\textbf{ERP} & Enterprise Resource Planning -- Hệ thống hoạch định nguồn lực doanh nghiệp \\
\hline
\textbf{EWMA} & Exponentially Weighted Moving Average -- Dự báo trung bình trượt có trọng số hàm mũ \\
\hline
\textbf{FEFO} & First-Expired, First-Out -- Nguyên tắc xuất bán ưu tiên lô cận hạn sử dụng trước \\
\hline
\textbf{FRN} & FreshRetailNet-50K -- Bộ dữ liệu nhu cầu hàng tươi sống thực có nhãn đứt hàng theo giờ \\
\hline
\textbf{GBDT} & Gradient Boosted Decision Trees -- Cây quyết định tăng cường độ dốc \\
\hline
\textbf{HITL} & Human-in-the-Loop -- Con người trong vòng lặp phê duyệt trước khi thực thi nghiệp vụ \\
\hline
\textbf{HPF} & High-Pass Filter -- Bộ lọc thông cao 85\,Hz pha không \\
\hline
\textbf{HSD} & Hạn sử dụng sản phẩm \\
\hline
\textbf{IQR} & Interquartile Range -- Khoảng tứ phân vị ($Q_3 - Q_1$) \\
\hline
\textbf{LightGBM} & Light Gradient Boosting Machine -- Thư viện cây tăng cường độ dốc dùng dự báo phân vị \\
\hline
\textbf{LLM} & Large Language Model -- Mô hình ngôn ngữ lớn \\
\hline
\textbf{LPU} & Language Processing Unit -- Bộ xử lý suy luận LLM phát luồng độ trễ thấp \\
\hline
\textbf{OOD / OOC} & Out-of-Distribution / Out-of-Catalog -- Đầu vào nằm ngoài phân phối hoặc ngoài danh mục \\
\hline
\textbf{OOS} & Out-of-Stock -- Tình trạng đứt hàng trên kệ \\
\hline
\textbf{OWASP} & Open Worldwide Application Security Project -- Chuẩn rủi ro bảo mật cho ứng dụng LLM \\
\hline
\textbf{PO} & Purchase Order -- Đơn đặt hàng nhập kho gửi nhà cung cấp \\
\hline
\textbf{POS} & Point of Sale -- Hệ thống máy tính tiền tại điểm bán \\
\hline
\textbf{RAG} & Retrieval-Augmented Generation -- Sinh văn bản tăng cường truy xuất tri thức \\
\hline
\textbf{SKU} & Stock-Keeping Unit -- Đơn vị mã hàng lưu kho \\
\hline
\textbf{SNR} & Signal-to-Noise Ratio -- Tỷ số tín hiệu trên nhiễu (đơn vị dB) \\
\hline
\textbf{SPO} & Smart ``Predict, then Optimize'' -- Khung dự báo định hướng quyết định tối ưu \\
\hline
\textbf{SSE} & Server-Sent Events -- Giao thức phát luồng phản hồi từng token \\
\hline
\textbf{SVD} & Singular Value Decomposition -- Phân rã giá trị kỳ dị chiếu không gian đặc trưng 896-d về 256-d \\
\hline
\textbf{TFLite} & TensorFlow Lite (LiteRT) -- Định dạng suy luận học sâu nhẹ trên thiết bị biên \\
\hline
\textbf{TTA} & Test-Time Augmentation -- Tăng cường đa góc nhìn tại thời điểm suy luận \\
\hline
\textbf{TTFT} & Time to First Token -- Độ trễ đến token hiển thị đầu tiên \\
\hline
\textbf{TTL} & Time to Live -- Thời gian sống tối đa của dữ liệu đệm / cầu nối \\
\hline
\textbf{TTS} & Text-to-Speech -- Tổng hợp tiếng nói từ văn bản \\
\hline
\textbf{VAD} & Voice Activity Detection -- Phát hiện khoảng tiếng nói (Silero VAD~v6) \\
\hline
\textbf{VLM} & Vision--Language Model -- Mô hình ngôn ngữ đa phương thức ảnh--văn bản \\
\hline
\textbf{VND} & Việt Nam Đồng \\
\hline
\textbf{WAPE} & Weighted Absolute Percentage Error -- Sai số phần trăm tuyệt đối có trọng số \\
\hline
\textbf{WER} & Word Error Rate -- Tỷ lệ lỗi từ trong nhận dạng tiếng nói \\
\hline
